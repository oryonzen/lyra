cmake_minimum_required(VERSION 3.15)
project(lyra C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

file(GLOB_RECURSE LYRA_SOURCES CONFIGURE_DEPENDS src/*.c)

add_library(lyra STATIC ${LYRA_SOURCES})

target_include_directories(lyra PUBLIC interface)

install(TARGETS lyra ARCHIVE DESTINATION lib)
install(DIRECTORY interface/ DESTINATION interface)
